//s branch//
const branchData = id => {
    const randomType = Math.floor(Math.random() * 3);
    let type;

    if(randomType === 0) {
        type = 'left';

    } else if(randomType === 1) {
        type = 'right';

    } else if(randomType === 2) {
        type = 'none'

    } else {
        return
    }
    
    let branchData = {id, type};

    return branchData
}
//e branch//

//s treepiece//
const crateTreePiece = branchData => {
    const branch = document.createElement('div');

    branch.className = 'trunk';
    branch.setAttribute('id', branchData.id);
    branch.innerHTML = `
        <div class="branch branch-${branchData.type}"></div>
    `;
    
    return branch
}
//e treepiece//

//s tree//
const treeElement = document.getElementById('tree');

let trunckNumID = 0;

const initialize = () => {
    for(let i = 0; i < 4; i++) {
        let trunk = crateTreePiece(branchData(i));
        
        trunckNumID = i;

        addPiece(trunk);
    }
}

const addPiece = piece => {
    treeElement.prepend(piece);
}

const removePiece = piece => {
    piece.remove();
}

const createPieceAfterChop = () => {
    addPiece(crateTreePiece(branchData(trunckNumID)));
} 

const controlTotalPiece = () => {
    let total = document.querySelectorAll('.trunk').length;
    let oldPiece = treeElement.firstElementChild;

    if(total >= 5) removePiece(oldPiece);
}

const getPositionBrunch = () => {
    let position = '';

    if (treeElement.lastElementChild.lastElementChild.classList.contains('branch-left')) {position = 'left';}
    else if (treeElement.lastElementChild.lastElementChild.classList.contains('branch-right')) {position = 'right';}
    else {position = 'none';}
    
    return position
}

let statusPositionjack;

const setPositionJack = (position) => {
    const jackPosition = document.getElementById('jack');
    
    if(position === 'left'){
        jackPosition.classList.remove('right');
        jackPosition.classList.add('left');

    } else if(position === 'right') {
        jackPosition.classList.remove('left');
        jackPosition.classList.add('right');

    } else return;

    statusPositionjack = position;
}

window.addEventListener('keydown', e1 => {
    if(gameOverStatus) return;

    if(
        (e1.key === "ArrowRight")
        ||
        (e1.key === "d")
    ) {
        setPositionJack('right');

    } else if(
        (e1.key === "ArrowLeft")
        ||
        (e1.key === "a")
    ) {
        setPositionJack('left');

    } else return;
});

const gameOverBox = document.getElementById('gameOver-box');

let gameOverStatus = false;

const gameOver = () => {
    if(getPositionBrunch() === statusPositionjack) {
        gameOverStatus = true;

        gameOverBox.classList.add('active');

//s gameplay//
const keys = ["ArrowRight", "ArrowLeft", "d", "a"];

window.onkeydown = e => {
    let lastTrunk = treeElement.lastElementChild;
    
    if(keys.includes(e.key)){
        if(gameOverStatus) return;
        
        if(lastTrunk) {
            removePiece(lastTrunk);
            gameOver();
            trunckNumID++;
            createPieceAfterChop();
            pointRating();
            tokenRating();
        }
    }
}

const pointRating = () => {
    if(gameOverStatus) return;

    scoreElement.textContent = `Score : ${score}`;
    scoreFlyElement.textContent = score;

    pop(scoreFlyElement);

    score++;

    if(!(Number(localStorage.getItem('bestScore')))) {
        localStorage.setItem('bestScore', bs);
    }

    if(Number(localStorage.getItem('bestScore'))) bs = Number(localStorage.getItem('bestScore'));

    if(score > bs) {
        bs = score;
        localStorage.setItem('bestScore', bs);
    }
}

const pop = (el) => {
    el.classList.remove('active');
    void el.offsetWidth;
    el.classList.add('active');
};

const gameOverBoxTokenTElement = document.getElementById('endGame-collectedT');

let tokenT = Number(localStorage.getItem('T')) ||  0;
let mst = 60;
let collectedTokenInGame = 0;

const tokenRating = () => {
    if((score % mst) === 0) {
        tokenT++;
        collectedTokenInGame++;
        mst += 60;

        localStorage.setItem('T', tokenT);

        tokenTElement.textContent = tokenT;
    }

    gameOverBoxTokenTElement.textContent = collectedTokenInGame;
}

const setTokenT = () => {
    let t = Number(localStorage.getItem('T'));
    
    tokenTElement.textContent = t;
}
